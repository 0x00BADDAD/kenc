Start -> Prog $

Prog -> Decls

Decls -> Decl Decls
Decls ->

Decl -> VarDec SEMI_COLON
Decl -> VarDecInit SEMI_COLON
Decl -> TypeDec SEMI_COLON
Decl -> FunDec

VarDecInit -> VAR ID ASSIGN Exp
VarDecInit -> VAR ID COLON TypeVal ASSIGN Exp
VarDec -> VAR ID
VarDec -> VAR ID COLON TypeVal

TypeDec -> TYPE ID ASSIGN TypeVal

TypeVal -> ID
TypeVal -> OPEN_BRACE TypeFields CLOSE_BRACE
TypeVal -> ARRAY OF TypeVal

TypeFields -> ID COLON TypeVal COMMA TypeFields
TypeFields -> ID COLON TypeVal

FunDec -> FUNC ID OPEN_PAREN TypeFields CLOSE_PAREN OPEN_BRACE Stmts CLOSE_BRACE
FunDec -> FUNC ID OPEN_PAREN TypeFields CLOSE_PAREN COLON TypeVal OPEN_BRACE Stmts CLOSE_BRACE


// Expression Tree

Exp -> LogicalOr

LogicalOr
    -> LogicalOr OR LogicalAnd
    | LogicalAnd

LogicalAnd
    -> LogicalAnd AND Disjunction
    | Disjunction

Disjunction
    -> Disjunction DIS Conjunction
    | Conjunction

Conjunction
    -> Conjunction CON Equality
    | Equality

Equality
    -> Equality ISEQUAL Rel
    | Equality NOT_EQUAL Rel
    | Rel

Rel
    -> Rel LESS_THAN Add
    |  Rel GREATER_THAN Add
    |  Rel LESS_EQUAL Add
    |  Rel GREATER_EQUAL Add
    |  Add

Add
    -> Add PLUS Mul
    | Add MINUS Mul
    | Mul

Mul
    -> Mul MUL Unary
    | Mul DIV Unary
    | Mul MOD Unary
    | Unary

Unary
    -> BOOL_NEGATION Unary
    | MINUS Unary
    | Primary

Primary
    -> NUM
    | STRING_LIT
    | Lvalue
    | FunCallExp
    | NIL
    | IfExpr
    | IfElseExpr
    | OPEN_PAREN Exp CLOSE_PAREN
    | RecordExpr
    | ArrayExpr
    | TRUE
    | FALSE


RecordExpr -> TypeVal OPEN_BRACE FieldInit CLOSE_BRACE
ArrayExpr -> TypeVal OPEN_SQUARE Exp CLOSE_SQUARE OF Exp

// currently empty records are erroneous

FieldInit
    ->ID ASSIGN Exp
    | ID ASSIGN Exp COMMA FieldInit

Stmts -> Stmt Stmts
Stmts ->

Stmt -> Assign SEMI_COLON
Stmt -> FunCallStmt SEMI_COLON
Stmt -> IfElseStmt
Stmt -> IfStmt
Stmt -> While
Stmt -> For
Stmt -> VarDec SEMI_COLON
Stmt -> VarDecInit SEMI_COLON
Stmt -> TypeDec SEMI_COLON
Stmt -> BreakStmt
Stmt -> ContinueStmt
Stmt -> Return

Return -> RETURN Exp SEMI_COLON
BreakStmt -> BREAK SEMI_COLON
ContinueStmt -> CONTINUE SEMI_COLON

FunCallStmt -> ID OPEN_PAREN FunArgs CLOSE_PAREN

FunArgs ->
FunArgs -> Exp ArgTail

ArgTail ->
ArgTail -> COMMA Exp ArgTail


Assign -> Lvalue ASSIGN Exp
Assign -> Lvalue ACC_DIV Exp
Assign -> Lvalue ACC_SUB Exp
Assign -> Lvalue ACC_MUL Exp
Assign -> Lvalue ACC_MOD Exp
Assign -> Lvalue ACC_PLUS Exp

Lvalue -> ID
Lvalue -> Lvalue ACCESSOR ID
Lvalue -> Lvalue OPEN_SQUARE Exp CLOSE_SQUARE

IfElseExpr -> IF OPEN_PAREN Exp CLOSE_PAREN OPEN_BRACE Stmts Exp SEMI_COLON CLOSE_BRACE ELSE OPEN_BRACE Stmts Exp SEMI_COLON CLOSE_BRACE

IfElseStmt -> IF OPEN_PAREN Exp CLOSE_PAREN OPEN_BRACE Stmts CLOSE_BRACE ELSE OPEN_BRACE Stmts CLOSE_BRACE


IfExpr -> IF OPEN_PAREN Exp CLOSE_PAREN OPEN_BRACE Stmts Exp SEMI_COLON CLOSE_BRACE

IfStmt -> IF OPEN_PAREN Exp CLOSE_PAREN OPEN_BRACE Stmts CLOSE_BRACE

While -> WHILE OPEN_PAREN Exp CLOSE_PAREN OPEN_BRACE Stmts CLOSE_BRACE

For -> FOR OPEN_PAREN ForFirsts SEMI_COLON ForSeconds SEMI_COLON ForThirds CLOSE_PAREN OPEN_BRACE Stmts CLOSE_BRACE

ForFirsts -> ForFirst COMMA ForFirsts
ForFirsts -> ForFirst

ForSeconds -> ForSecond COMMA ForSeconds
ForSeconds -> ForSecond

ForThirds -> ForThird COMMA ForThirds
ForThirds -> ForThird

ForFirst -> VarDecInit
ForFirst -> Assign
ForFirst ->

ForSecond -> Exp
ForSecond ->

ForThird -> Assign
ForThird ->
